function requestApi(options = {}) {
  const { url, method = 'GET', data = {}, success, fail, showLoading = true } = options;
  
  if (showLoading) {
    wx.showLoading({ title: '请求中...', mask: true });
  }
  
  wx.request({
    url: BASE_URL + url,
    method,
    data,
    header: {
      'Content-Type': 'application/json',
      'Authorization': getAuthToken()  // 自动添加认证头
    },
    success: (res) => {
      if (showLoading) {
        wx.hideLoading();
      }
      
      // 处理 401 未授权
      if (res.statusCode === 401) {
        wx.removeStorageSync('currentUser');
        wx.reLaunch({ url: '/pages/login/login' });
        return;
      }
      
      if (typeof success === 'function') {
        success(res);
      }
    },
    fail: (err) => {
      if (showLoading) {
        wx.hideLoading();
      }
      showError('网络请求失败，请检查网络或稍后重试');
      if (typeof fail === 'function') {
        fail(err);
      }
    }
  });
}

```javascript
function validateInput(input) {
  if (input === null || input === undefined) {
    return { valid: false, number: null, error: '请输入数字' };
  }
  
  const str = String(input);
  const trimmed = str.trim();
  
  if (!trimmed) {
    return { valid: false, number: null, error: '请输入数字' };
  }
  
  // 只允许 0-9
  const cleaned = trimmed.replace(/[^0-9]/g, '');
  if (!cleaned) {
    return { valid: false, number: null, error: '请输入数字（0-9）' };
  }
  
  // 检查前导零
  if (cleaned.length > 1 && cleaned[0] === '0') {
    return { valid: false, number: null, error: '数字不能有前导零' };
  }
  
  // 位数限制（最多 200 位）
  if (cleaned.length > 200) {
    return { valid: false, number: null, error: '数字位数不能超过200位' };
  }
  
  return { valid: true, number: cleaned, error: null };
}
```

**说明**：
- 只做前端输入格式验证
- 不进行实际分解计算（计算在后端）
- 返回验证结果和清理后的数字字符串

## 核心算法详解

### 1. Miller-Rabin 素性测试

**算法原理**：

Miller-Rabin 是一种概率性素性测试算法，基于费马小定理的推广。

**实现代码**：

```python
def miller_rabin(n, k=5):
    """
    Miller-Rabin素性测试
    :param n: 待测试的数
    :param k: 测试轮数，默认5轮
    :return: True表示可能是素数，False表示合数
    """
    # 对于大数字，自动增加测试轮数
    num_digits = len(str(n))
    if num_digits > 100 and k < 10:
        k = 10
    elif num_digits > 150 and k < 15:
        k = 15
    
    if n == 2 or n == 3:
        return True
    if n < 2 or n % 2 == 0:
        return False
    
    # 将 n-1 写成 d * 2^r 的形式
    d = n - 1
    r = 0
    while d % 2 == 0:
        d = d // 2
        r += 1
    
    # 进行k轮测试
    for i in range(k):
        a = random.randint(2, n - 2)
        x = mod_pow(a, d, n)
        
        if x == 1 or x == n - 1:
            continue
        
        composite = True
        for j in range(r - 1):
            x = (x * x) % n
            if x == n - 1:
                composite = False
                break
        
        if composite:
            return False  # 确定是合数
    
    return True  # 可能是素数
```

**时间复杂度**：O(k log³ n)，其中 k 是测试轮数

**准确性**：进行 k 轮测试后，错误概率 ≤ 4^(-k)

### 2. Pollard's Rho 算法

**算法原理**：

Pollard's Rho 是一种用于分解合数的随机算法，基于生日悖论和 Floyd 循环检测。

**实现代码**：

```python
def pollard_rho(n):
    """
    Pollard's Rho算法（Brent变体）
    :param n: 待分解的数
    :return: 返回一个因子，如果失败返回None
    """
    if n % 2 == 0:
        return 2
    if miller_rabin(n):
        return None  # 如果是素数，无法分解
    
    # 根据数字大小动态调整最大迭代次数
    num_digits = len(str(n))
    if num_digits <= 20:
        max_iterations = 100000
    elif num_digits <= 40:
        max_iterations = 500000
    elif num_digits <= 50:
        max_iterations = 1000000
    # ... 更多条件
    
    # 使用随机初始值
    x = random.randint(1, n - 1)
    y = x
    k = 2
    
    for i in range(max_iterations):
        x = (x * x - 1) % n
        d = gcd(abs(y - x), n)
        
        if d > 1 and d < n:
            return d
        
        if i == k:
            y = x
            k = k * 2
    
    return None
```

**时间复杂度**：平均 O(√p)，其中 p 是最小因子

**优化**：
- 使用 Brent 变体提高效率
- 根据数字大小动态调整迭代次数
- 先进行试除法找小因子

### 3. 完整分解流程

```python
def factorize(n, mode='standard'):
    """
    完整的大整数分解函数
    :param n: 待分解的数
    :param mode: 分解模式（'standard' 或 'fast'）
    :return: 因子列表
    """
    factors = []
    
    # 处理特殊情况
    if n == 0:
        return [0]
    if n == 1:
        return [1]
    
    # 先进行试除法找小因子
    while True:
        trial_factor = trial_division(n)
        if trial_factor:
            factors.append(trial_factor)
            n = n // trial_factor
            if n == 1:
                break
        else:
            break
    
    # 如果还有剩余部分，使用 Pollard's Rho
    if n > 1:
        if miller_rabin(n):
            factors.append(n)
        else:
            # 递归分解
            d = pollard_rho(n)
            if d:
                factors.extend(factorize(d, mode))
                factors.extend(factorize(n // d, mode))
            else:
                factors.append(n)  # 无法分解，可能是大素数
    
    return sorted(factors)
```

## 关键代码片段

### 1. 登录验证装饰器

```python
def login_required(f):
    """登录验证装饰器"""
    @wraps(f)
    def decorated_function(*args, **kwargs):
        token = request.headers.get('Authorization')
        if not token:
            return jsonify({'success': False, 'message': '未登录'}), 401
        
        user = User.query.filter_by(username=token).first()
        if not user:
            return jsonify({'success': False, 'message': '无效的 token'}), 401
        
        request.current_user = user
        return f(*args, **kwargs)
    return decorated_function
```

**说明**：
- 从请求头获取 Token（用户名）
- 查询用户并附加到 `request.current_user`
- 如果验证失败，返回 401 错误

### 2. 管理员权限装饰器

```python
def admin_required(f):
    """管理员权限装饰器"""
    @wraps(f)
    @login_required
    def wrapper(*args, **kwargs):
        user = request.current_user
        if user.level != 'admin':
            return jsonify({
                'success': False,
                'message': '需要管理员权限'
            }), 403
        return f(*args, **kwargs)
    return wrapper
```

**说明**：
- 先验证登录（`@login_required`）
- 再检查用户等级是否为管理员
- 使用 `@wraps` 保留原函数名，避免 Flask 端点冲突

    ax.set_xlabel('时间（小时）')
    ax.set_ylabel('使用次数')
    ax.set_title('不同职业的使用时间分布')
    ax.legend()
    ax.grid(True, alpha=0.3)
    
    # 保存为 Base64
    img_buffer = io.BytesIO()
    plt.savefig(img_buffer, format='png', dpi=100)
    img_buffer.seek(0)
    img_base64 = base64.b64encode(img_buffer.read()).decode()
    plt.close()
    
    return jsonify({
        'success': True,
        'data': {
            'image': f'data:image/png;base64,{img_base64}',
            'jobs': jobs,
            'hours': hours,
            'data': {job: [job_hour_stats[job][h] for h in hours] for job in jobs}
        }
    })
